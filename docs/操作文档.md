### 2.2 写 `models.py`（严格按 API.md 第 4 节）

**操作**：新建 `models.py`，内容如下：

```python
from typing import List, Literal, Optional
from pydantic import BaseModel, Field


class KnowledgeBlock(BaseModel):
    id: str
    name: str
    summary: str
    source_excerpt: str
    scope: Optional[str] = ""


class GenerateRequest(BaseModel):
    knowledge_blocks: List[KnowledgeBlock] = Field(..., min_length=1, max_length=20)
    question_types: List[Literal["single", "blank"]]
    count: int = Field(..., ge=15, le=20)
    existing_stems: Optional[List[str]] = Field(default_factory=list)


class Question(BaseModel):
    id: str
    type: Literal["single", "blank"]
    stem: str
    options: List[str]  # 单选 4 项且已带 "A."/"B." 前缀；填空为 []
    answer: str
    explanation: str
    knowledge_point_id: str
    source_excerpt: str
    cognitive_level: Literal["记忆", "理解", "应用", "分析"]
    steps: int
    difficulty: Literal["简单", "中等"]


class GenerateResponse(BaseModel):
    fallback: bool
    questions: List[Question]
    error: Optional[dict] = None


class ExtractRequest(BaseModel):
    text: str = Field(..., min_length=100, max_length=1000)


class ExtractResponse(BaseModel):
    fallback: bool
    knowledge_blocks: List[KnowledgeBlock]
    error: Optional[dict] = None
```

**预期**：文件无语法错误；题型只有 `single/blank`，填空 `options` 用空数组，`difficulty` 只有“简单/中等”，两个 Response 都有 `error` 可选字段。

**自检**：
- [ ] `python -c "from models import KnowledgeBlock, GenerateRequest, Question, GenerateResponse, ExtractRequest, ExtractResponse; print('models ok')"` 输出 `models ok`
- [ ] 文件里没有 `single_choice`、`fill_blank`、`困难`
- [ ] `GenerateResponse` 和 `ExtractResponse` 都有 `error: Optional[dict] = None`

```python
FAKE_QUESTIONS = build_fake_questions()


@app.post("/generate", response_model=GenerateResponse)
def generate(req: GenerateRequest):
    # 今天不接真实 LLM，直接返回假题目；knowledge_point_id 动态替换为请求中的第一个知识块 id
    # 假数据阶段只按题型过滤，不按知识块轮转；真实生成再按知识块轮转
    kid = req.knowledge_blocks[0].id
    pool = [q for q in FAKE_QUESTIONS if q.type in req.question_types]
    selected = [
        q.model_copy(update={"knowledge_point_id": kid})
        for q in pool[: req.count]
    ]

    # 仅假数据阶段存在；明天接真实 LLM 后移除，不写入 API.md 错误码表
    error = None
    if len(selected) < req.count:
        error = {
            "code": "INSUFFICIENT_FAKE_QUESTIONS",
            "message": f"假数据池按题型过滤后仅 {len(selected)} 道，少于请求的 {req.count} 道；接真实 LLM 后消除",
        }

    return GenerateResponse(fallback=False, questions=selected, error=error)


@app.post("/extract", response_model=ExtractResponse)
def extract(_req: ExtractRequest):
    # P1 占位：返回符合 ExtractResponse 的结构，HTTP 200
    # fallback=False 表示未使用预置结果；前端按 error.code 判断未实现
    return ExtractResponse(
        fallback=False,
        knowledge_blocks=[],
        error={"code": "NOT_IMPLEMENTED", "message": "P1 功能，占位接口，未使用预置结果"},
    )
```

### 2.6 写 `test_generate.py` 自动自检

**操作**：新建 `test_generate.py`：

```python
import json
import sys

import requests

BASE = "http://127.0.0.1:8000"

with open("request_example.json", "r", encoding="utf-8") as f:
    base_payload = json.load(f)


def run_case(count: int):
    payload = dict(base_payload)
    payload["count"] = count

    r = requests.post(f"{BASE}/generate", json=payload, timeout=30)
    assert r.status_code == 200, r.text
    data = r.json()

    assert data["fallback"] is False
    qs = data["questions"]

    single = [q for q in qs if q["type"] == "single"]
    blank = [q for q in qs if q["type"] == "blank"]
    assert 12 <= len(single) <= 15, f"count={count} 单选数量异常: {len(single)}"
    assert 3 <= len(blank) <= 5, f"count={count} 填空数量异常: {len(blank)}"
    assert len(qs) == count or data.get("error"), f"count={count} 题目数量异常: {len(qs)}"

    for q in qs:
        assert q["id"].startswith("q_"), q["id"]
        assert q["stem"], q
        assert q["explanation"], q
        assert q["source_excerpt"], q
        assert q["knowledge_point_id"] == payload["knowledge_blocks"][0]["id"], q
        assert q["difficulty"] in ("简单", "中等"), q
        assert q["cognitive_level"] in ("记忆", "理解", "应用", "分析"), q
        assert isinstance(q["steps"], int) and q["steps"] >= 1, q

        if q["type"] == "single":
            assert len(q["options"]) == 4, q
            assert q["answer"] in ("A", "B", "C", "D"), q
            for idx, prefix in enumerate(("A.", "B.", "C.", "D.")):
                assert q["options"][idx].startswith(prefix), q
        else:
            assert q["options"] == [], q
            assert q["answer"], q

    print(f"OK: count={count}, 总数={len(qs)}, 单选={len(single)}, 填空={len(blank)}")


try:
    for c in (15, 18, 20):
        run_case(c)
except requests.exceptions.ConnectionError:
    print("连接失败：请先启动 uvicorn main:app --reload --host 127.0.0.1 --port 8000", file=sys.stderr)
    sys.exit(1)

print("OK: /generate 自检全部通过")
```

```text
OK: count=18, 总数=18, 单选=15, 填空=3
OK: count=20, 总数=20, 单选=15, 填空=5
OK: /generate 自检全部通过
```
